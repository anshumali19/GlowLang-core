#define R(idx) vm_stack[base_reg + (idx)]
#define DISPATCH() { \
    if (profiling_active) pc->exec_count++; \
    if (has_exception) goto l_OP_THROW; \
    fprintf(stdout, "OP: %d\\n", pc->op); fflush(stdout); \
    goto *dt[pc->op]; \
}
    DISPATCH();

    l_OP_CONST_INT: R(pc->r0) = value_int(pc->arg); pc++; DISPATCH();
    l_OP_CONST_FLOAT: R(pc->r0) = *(Value*)pc->ptr; pc++; DISPATCH();
    l_OP_CONST_STRING: R(pc->r0) = value_string_copy((const char*)pc->ptr); pc++; DISPATCH();
    l_OP_CONST_BOOL: R(pc->r0) = value_bool(pc->arg); pc++; DISPATCH();
    l_OP_LOAD_VAR: {
        int slot = pc->arg;
        const char *name = (const char*)pc->ptr;
        
        // 0. Thread-local self Path
        if (name && strcmp(name, "self") == 0) {
            R(pc->r0) = current_self;
            pc++; DISPATCH();
        }

        // 1. Inline Cache Path (Fast)
        if (current_closure && pc->cache[0].key == current_closure) {
            R(pc->r0) = current_closure->captured_values[pc->cache[0].val];
            pc++; DISPATCH();
        }

        // 2. Upvalue Path (Slow resolution)
        if (current_closure && name) {
            for (int i = 0; i < current_closure->captured_count; i++) {
                if (strcmp(current_closure->captured_names[i], name) == 0) {
                    pc->cache[0].key = current_closure;
                    pc->cache[0].val = i;
                    R(pc->r0) = current_closure->captured_values[i];
                    pc++; DISPATCH();
                }
            }
        }

        // 3. Global Table Path (Legacy)
        if (slot >= global_table.count) { fprintf(stderr, "ERROR: LOAD_VAR slot %d out of bounds (%d)\n", slot, global_table.count); exit(1); }
        Value val = global_table.values[slot];
        if (GET_TYPE(val) == VAL_NONE && name) {
            FuncEntry *fe = func_get(name);
            if (fe) {
                val = bound_method_new(value_none(), fe);
                global_table.values[slot] = val; // Cache it
            }
        }
        R(pc->r0) = val;
        pc++; DISPATCH();
    }
    l_OP_STORE_VAR: {
        int slot = pc->arg;
        const char *name = (const char*)pc->ptr;

        // 0. Thread-local self Path
        if (name && strcmp(name, "self") == 0) {
            current_self = R(pc->r0);
            pc++; DISPATCH();
        }

        // 1. Inline Cache Path (Fast)
        if (current_closure && pc->cache[0].key == current_closure) {
            gc_write_barrier((GCObject*)current_closure, R(pc->r0));
            current_closure->captured_values[pc->cache[0].val] = R(pc->r0);
            pc++; DISPATCH();
        }

        // 2. Upvalue Path (Slow resolution)
        if (current_closure && name) {
            for (int i = 0; i < current_closure->captured_count; i++) {
                if (strcmp(current_closure->captured_names[i], name) == 0) {
                    pc->cache[0].key = current_closure;
                    pc->cache[0].val = i;
                    gc_write_barrier((GCObject*)current_closure, R(pc->r0));
                    current_closure->captured_values[i] = R(pc->r0);
                    pc++; DISPATCH();
                }
